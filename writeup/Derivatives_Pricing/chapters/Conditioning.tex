\subsection{Conditional Expectation}
In elementary probability, we define the conditional expectation after defining conditional probability. In advanced probability, it is more convenient to reverse the order, as we often want to condition on $\sigma$-algebras. In what follows, $(\Omega, \cH, \Prob)$ is a probability space, $X$ a random variable, and $\cF$ a sub-$\sigma$-algebra of $\cH$; that is a $\sigma$-algebra whose every member (i.e., set) belongs to $\cH$ as well. Recall the notation $V \in \cF_+$ to mean that $V \geq 0$ and is $\cF$-measurable.

\begin{defn}{Conditional Expectation}{ConditionalExpectation}
The conditional expectation of $X$ given $\cF$, denoted $\E_{\cF} X$, is defined in two steps:
\begin{enumerate}
    \item For $X \ge 0$, $\E_{\cF} X$ is defined as any random variable $\hat{X}$ with
    \begin{enumerate}
        \item $\hat{X} \in \cF_+$.
        \item $\E V X = \E V \hat{X}$ for every $V \in \cF_+$ (\textit{projection property}).
    \end{enumerate}
    \item For arbitrary $X$, if $\E X$ exists (possibly $\pm \infty$), $\E_{\cF} X \coloneqq \E_{\cF} X^+ - \E_{\cF} X^-$. Otherwise, the conditional expectation is left undefined.
\end{enumerate}
\end{defn}

Note that the conditional expectation is defined \textit{almost surely}.

\begin{thm}{Existence and Uniqueness of the Conditional Expectation}{}
Let $X \ge 0$ be a random variable and let $\cF$ be a sub-$\sigma$-algebra of $\cH$. Then, $\E_{\cF} X$ exists and is unique up to equivalence.
\end{thm}

\begin{thm}{Additional Properties of the Conditional Expectation}{}
Let $\cF$ and $\mathcal{G}$ be sub-$\sigma$-algebras of $\cH$. Let $W$ and $X$ be random variables such that $\E X$ and $\E WX$ exist. Then, the following hold:
\begin{enumerate}
    \item (\textit{Taking out what is known}): If $W \in \cF$, then $\E_{\cF} WX = W \E_{\cF} X$.
    \item (\textit{Repeated conditioning}): If $\cF \subseteq \mathcal{G}$, then
    \begin{equation}
        \E_{\cF} \E_{\mathcal{G}} X = \E_{\mathcal{G}} \E_{\cF} X = \E_{\cF} X.
        \label{eq:TowerProperty}
    \end{equation}
\end{enumerate}
\end{thm}

There is an amusing way to remember repeated conditioning: the $\sigma$-algebras $\cF$ and $\cG$ represent the information about the measurement on a random experiment by a fool and a genius. The fool has no use for the genius' information,
\[
    \E_{\cF}\E_{\cG} X = \E_{\cF} X,
\]
and, sadly, the genius cannot further improve on the poor information of the fool,
\[
    \E_{\cG}\E_{\cF} X = \E_{\cF} X.
\]

In many applications of conditioning, we wish to take the expectation of a random variable $X$ conditional on one or more random variables or, in general, on a stochastic process $Y \coloneqq \{Y_t, t \in \bb{T}\}$, in which case $\cF = \sigma(Y) = \sigma\{Y_t, t \in \bb{T}\}$. In this case we often write
\[
    \E_{\cF} X = \E[X \gvn Y_t, t \in \bb{T}] = \E[X \gvn Y].
\]

When conditioning on a random variable $Y$, we can use the fact (see \Cref{thm:SigmaXNumericalRandomVariables}) that any random variable in $\sigma(Y)$, in particular $\E[X \gvn Y]$, must be a deterministic measurable function of $Y$; that is, $\E[X \gvn Y] = h(Y)$ for some measurable function $h$. This is in accordance with our use of conditional expectations in elementary probability: we first determine $h(y) \coloneqq \E[X \gvn Y = y]$ and then use $\E[X \gvn Y] = h(Y)$ in computations.


\begin{thm}{Conditional Expectation in $L^2$}{}
For $X \in L^2$, the conditional expectation $\E_{\cF} X$ is the orthogonal projection of $X$ onto the subspace $\mathcal{V}$ of $\cF$-measurable random variables in $L^2$.
\end{thm}

\subsection{Conditional Probability and Distribution}

\begin{defn}{Conditional Probability for Events}{ConditionalProbabilityEvents}
Let $A$ and $B$ be two events with $\Prob(B) > 0$. The \textit{conditional probability} that $A$ occurs given that $B$ occurs is denoted by
\begin{equation}
    \Prob(A \mid B) \coloneqq \frac{\Prob(A \cap B)}{\Prob(B)}.
    \label{eq:ConditionalProbabilityDef}
\end{equation}
\end{defn}

\begin{thm}{Properties Involving Conditional Probabilities}{}
Let $(\Omega, \cH, \Prob)$ be a probability space and $A, A_1, A_2, \dots, A_n, B_1, B_2, \dots$ events such that $\Prob(A_1 \cap \dots \cap A_n) > 0$ and $\{B_i\}$ form a partition of $\Omega$, with $\Prob(B_i) > 0$ for all $i$. Then, the following hold:
\begin{enumerate}
    \item \textit{Law of total probability}: $\Prob(A) = \sum_i \Prob(A \mid B_i) \Prob(B_i)$.
    \item \textit{Bayes' rule}: $\Prob(B_j \mid A) = \frac{\Prob(B_j) \Prob(A \mid B_j)}{\sum_i \Prob(B_i) \Prob(A \mid B_i)}$.
    \item \textit{Chain or product rule}:
    \begin{equation*}
        \Prob(\cap_{i=1}^n A_i) = \Prob(A_1) \Prob(A_2 \mid A_1) \cdots \Prob(A_n \mid A_1 \cap \dots \cap A_{n-1}).
    \end{equation*}
\end{enumerate}
\end{thm}

\begin{defn}{Conditional Distribution}{}
Let $Y$ be a random variable on the probability space $(\Omega, \cH, \Prob)$ taking values in the measurable space $(E, \cE)$. Let $\cF$ be a sub-$\sigma$-algebra of $\cH$. The \textit{conditional distribution of $Y$ given $\cF$} is any probability transition kernel $L$ from $(\Omega, \cF)$ to $(E, \cE)$ such that
\begin{equation}
    \Prob_{\cF}(Y \in B) = L(\cdot, B), \quad B \in \cE.
    \label{eq:ConditionalDistributionDef}
\end{equation}
\end{defn}

\begin{thm}{Existence of the Conditional Distribution}{}
For every standard measurable space $(E, \cE)$ there exists a version of the conditional distribution of $Y$ given $\cF$.
\end{thm}

\begin{thm}{Regular Version of a Conditional Probability}{RegularVersion}
    Let $(\Omega, \cH)$ be a standard measurable space and $\cF$ a sub-$\sigma$-algebra of $\cH$. Then, there \emph{exists} a regular version of the conditional probability $\Pm_{\cF}$.
\end{thm}

\begin{equation}\label{eq:PiForm}
    \pi(\di x, \di y) = \mu(\di x)K(x, \di y).
\end{equation}

\begin{thm}{Conditional Distribution}{ConditionalDistribution}
    If $\pi$ is of the form \cref{eq:PiForm}, then the kernel $L$ defined by
    \begin{equation}\label{eq:KernelDefinition}
        L(\omega, B) \coloneqq K(X(\omega), B), \quad \omega \in \Omega, B \in \cE
    \end{equation}
    is a version of the conditional distribution of $Y$ given $X$.
\end{thm}

\begin{thm}{Disintegration}{Disintegration}
    Let $\pi$ be a probability measure on the product space $(D \times E, \c{D} \otimes \c{E})$, where $(E, \c{E})$ is a standard measurable space. Then, there exists a probability measure $\mu$ on $(D, \c{D})$ and a transition kernel $K$ from $(D, \c{D})$ to $(E, \c{E})$ such that
    \begin{equation}\label{eq:DisintegrationFormula}
        \pi(\di x, \di y) = \mu(\di x) K(x, \di y), \quad x \in D, y \in E.
    \end{equation}
\end{thm}

\begin{defn}{Conditional Independence}{ConditionalIndependence}
    Let $\cF, \cF_1, \dots, \cF_n$ be sub-$\sigma$-algebras of $\cH$. The $\{\cF_i\}$ are said to be \emph{conditionally independent} given $\cF$ if
    \begin{equation}\label{eq:ConditionalIndependenceDef}
        \E_{\cF} V_1 \dots V_n = \E_{\cF} V_1 \dots \E_{\cF} V_n
    \end{equation}
    for all positive random variables $V_1, \dots, V_n$ in $\cF_1, \dots, \cF_n$, respectively.
\end{defn}

\begin{prop}{Conditional Independence}{ConditionalIndependenceProp}
    $\cF_1$ and $\cF_2$ are conditionally independent given $\cF$ if and only if
    \begin{equation}\label{eq:ConditionalIndependenceProp}
        \E_{\cF \vee \cF_1} V_2 = \E_{\cF} V_2 \quad \text{for all positive } V_2 \in \cF_2.
    \end{equation}
\end{prop}

\begin{equation}\label{eq:ITConsistencyCondition}
    \Pm(B_0 \times \cdots \times B_n \times E_{n+1} \times \cdots) = \pi_n(B_0 \times \cdots \times B_n), \quad B_i \in \cE_i, i = 0, \dots, n.
\end{equation}

\subsection{Existence of Probability Spaces}


\begin{thm}{Ionescu--Tulcea's Existence Theorem}{IonescuTulceasExistenceTheorem}
    There exists a unique probability measure $\Pm$ on $(\Omega, \cH)$ such that \cref{eq:ITConsistencyCondition} holds.
\end{thm}

\begin{thm}{Kolmogorov's Extension Theorem}{KolmogorovsExtensionTheorem}
    Let $(E, \cE)$ be a standard measurable space and $K$ an index set. For each finite subset $J$ of $K$, let $\pi_J$ be a probability measure on the product space $(E^J, \cE^J)$. If the $(\pi_J)$ are \emph{consistent} in the sense that
    \begin{equation*}
        \pi_I = \pi_J \circ \mathrm{proj}_{JI}^{-1}, \quad I \subseteq J, \quad |J| < \infty,
    \end{equation*}
    then there exists a unique probability measure $\Pm$ on $(\Omega, \cH)$ and a stochastic process $X$ taking values in $(E, \cE)^K$ such that, with $X_J \coloneqq (X_j)_{j \in J}$:
    \begin{equation*}
        \Pm(X_J \in A) = \pi_J(A), \quad A \in \cE^J, \quad |J| < \infty.
    \end{equation*}
\end{thm}

\begin{defn}{Hölder Continuity}{HolderContinuity}
    A function $f : \R_+ \to \R$ is said to be \emph{Hölder continuous} of order $\alpha > 0$ on $B \subseteq \R_+$ if there is a constant $k$ such that
    \begin{equation*}
        |f(t) - f(s)| \leq k |t - s|^\alpha, \quad s, t \in B.
    \end{equation*}
\end{defn}

\begin{thm}{Hölder Continuous Modification}{HolderContinuousModification}
    Let $X \coloneqq (X_t, t \in [0, 1])$ be a stochastic process with state space $\R$. If there exist constants $c, p, q$ in $(0, \infty)$ such that
    \begin{equation}\label{eq:KolmogorovChentsovCondition}
        \E|X_t - X_s|^p \leq c |t - s|^{1+q}, \quad s, t \in [0, 1],
    \end{equation}
    then for every $\alpha \in (0, q/p)$ there is a modification $\widetilde{X}$ of $X$ that is almost surely Hölder continuous of order $\alpha$ on $[0, 1]$.
\end{thm}

\subsection{Markov Property}
One of the most studied stochastic processes is the Markov process. In a way it can be viewed as the probabilistic analogue of a difference or differential equation, in the sense that, conditional on the past history, future increments depend only on the present state. The concepts of conditional probability and expectation are thus essential to the analysis.

Throughout this section, $\bb{T}$ is some subset of $\R$ and $X \idef (X_t, t \in \bb{T})$ is a stochastic process on some probability space $(\Omega, \cH, \Prob)$, taking values in some measurable space $(E, \cE)$.

\begin{defn}{Adaptedness}{Adaptedness}
    A stochastic process $X \coloneqq (X_t, t \in \bb{T})$ taking values in some measurable space $(E, \cE)$ is said to be \emph{adapted} to a filtration $\cF \coloneqq (\cF_t, t \in \bb{T})$ if for all $t \in \bb{T}$, $X_t$ is $\cF_t/\cE$-measurable.
\end{defn}

In what follows, we assume that the process $X$ is adapted to some filtration $\F \idef (\F_t, t \in \bb{T})$ --- for example, the natural filtration of $X$, in which case $\F_t \idef \sigma(X_u, u \leq t, u \in \bb{T})$. Let $\cG_t \idef \sigma(X_u, u \geq t, u \in \bb{T})$ be the \textit{future} $\sigma$-algebra of $X$ for time $t$.

\begin{defn}{Markov Property}{MarkovProperty}
    $X$ is said to be \emph{Markovian} relative to $\cF$ if for every $t$, the past $\cF_t$ and the future $\cG_t$ are conditionally independent, given the present state $X_t$.
\end{defn}

\begin{defn}{Markovian Transition Function}{MarkovianTransitionFunction}
    A \emph{Markovian transition function} on $(E, \cE)$ is a family $(P_{t,u}, t, u \in \bb{T}, t < u)$ of probability transition kernels on $(E, \cE)$ that satisfy the \emph{Chapman--Kolmogorov equations}:
    \begin{equation}\label{eq:ChapmanKolmogorov}
        P_{s,t}P_{t,u} = P_{s,u}, \quad s < t \leq u.
    \end{equation}
\end{defn}

A Markovian process $X$ is said to admit $(P_{t,u})$ as a transition function if
\[
    \E[f(X_u) \gvn X_t] = (P_{t,u}f)(X_t), \quad t < u, \ f \in \cE_+.
\]
For $f = \I_A$ with $A \in \cE$, this means
\[
    \Prob(X_u \in A \gvn X_t = x) = P_{t,u}(x, A),
\]
giving an intuitive meaning to the transition functions.

Define $P_t \idef P_{0,t}$. If $P_{t,u} = P_{u-t}$ for all $t \leq u$, the associated Markovian process $X$ is said to be \textit{time-homogeneous} and $P_t$ satisfies the \textit{semi-group property}
\[
    P_{t+s} = P_s P_t, \quad s, t \geq 0.
\]

Markov processes can be classified according to the nature (discrete or continuous) of their state space $E$ and time set $\bb{T}$. We call a Markovian process a \textit{Markov chain}\footnote{There is no consensus on this nomenclature. Some call a Markovian process a Markov chain if $E$ is discrete.} if $\bb{T}$ is discrete. The Markov process \textit{par excellence} in continuous time is the \Levy{} process; this includes the Wiener, Poisson, compound Poisson, and pure-jump \Levy{} processes.

\begin{thm}{Generator of a Markov Jump Process}{GeneratorOfAMarkovJumpProcess}
    The infinitesimal generator of a Markov jump process with Lévy kernel $L$ is given by
    \begin{equation}\label{eq:GeneratorMarkovJump}
        (Gf)(x) = \int_{\R^d} L(x, \di y)[f(y) - f(x)], \quad f \in \mathcal{B}^d(\mathrm{bounded}).
    \end{equation}
\end{thm}

\begin{prop}{Poisson Integrals}{PoissonIntegrals}
    Let $N$ be a Poisson random measure on $\R_+ \times E$ with mean measure $\mu$ satisfying $\mu(\{0\} \times E) = 0$. Let $R \coloneqq (R(t, z), t \geq 0, z \in E)$ be a positive stochastic process in $\cF^P \otimes \cE$. Then,
    \begin{equation}\label{eq:PoissonIntegralsExpectation}
        \E N R = \E \mu R.
    \end{equation}
\end{prop}
