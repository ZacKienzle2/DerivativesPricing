\subsection{Asset Price Modelling}
\noindent The prices at time $t \ge 0$ of $d+1$ assets numbered $0, 1, \dots, d$ is denoted by the random vector
\begin{equation} \label{eq:SpotPriceVector}
	\xbar{S}_t = \left(S_t^{(0)}, S_t^{(1)}, \dots, S_t^{(d)}\right)
\end{equation}
which forms a stochastic process $(\xbar{S}_t)_{t \in \R_+}$. As in discrete time, the asset $n^o$ 0 is a riskless asset (of savings account type) yielding an interest rate $r$, \textit{i.e.} we have
\begin{equation} \label{eq:RisklessAssetContinuous}
	S_t^{(0)} = S_0^{(0)} \e^{rt}, \quad t \ge 0.
\end{equation}

\begin{defn}{Discounting}{DiscountingDef}
	Let
	\begin{equation} \label{eq:DiscountedVectorDef}
		\xbar{X}_t \coloneqq \left(\widetilde{S}_t^{(0)}, \widetilde{S}_t^{(1)}, \dots, \widetilde{S}_t^{(d)}\right), \quad t \in \R,
	\end{equation}
	denote the vector of discounted asset prices, defined as:
	\begin{equation} \label{eq:DiscountedComponentDef}
		\widetilde{S}_t^{(k)} = \e^{-rt}S_t^{(k)}, \quad t \ge 0, \quad k = 0, 1, \dots, d.
	\end{equation}
\end{defn}

\noindent We can also write
\begin{equation} \label{eq:DiscountedVectorCompact}
	\xbar{X}_t \coloneqq \e^{-rt}\xbar{S}_t, \quad t \ge 0.
\end{equation}

\begin{defn}{Portfolio Strategy}{PortfolioStrategyDef}
	A portfolio strategy is a stochastic process $(\xbar{\xi}_t)_{t \in \R_+} \subset \R^{d+1}$, where $\xi_t^{(k)}$ denotes the (possibly fractional) quantity of asset $n^o$ $k$ held at time $t \ge 0$.
\end{defn}

\noindent The value at time $t \ge 0$ of the portfolio strategy $(\xbar{\xi}_t)_{t \in \R_+} \subset \R^{d+1}$ is defined by
\begin{equation} \label{eq:PortfolioValueDef}
	V_t \coloneqq \xbar{\xi}_t \cdot \xbar{S}_t = \sum_{k=0}^d \xi_t^{(k)} S_t^{(k)}, \quad t \ge 0.
\end{equation}

\noindent The discounted value of the portfolio is defined by
\begin{align}
	\widetilde{V}_t & \coloneqq \e^{-rt} V_t \label{eq:DiscountedValueDef}                              \\
	                & = \e^{-rt} \xbar{\xi}_t \cdot \xbar{S}_t \nonumber                                \\
	                & = \e^{-rt} \sum_{k=0}^d \xi_t^{(k)} S_t^{(k)} \nonumber                           \\
	                & = \sum_{k=0}^d \xi_t^{(k)} \widetilde{S}_t^{(k)} \nonumber                        \\
	                & = \xbar{\xi}_t \cdot \xbar{X}_t, \quad t \ge 0. \label{eq:DiscountedValueCompact}
\end{align}

\noindent The effect of discounting from time $t$ to time $0$ is to divide prices by $\e^{rt}$, making all prices comparable at time $0$.

\subsection{Arbitrage and Risk-Neutral Measures}
\noindent In continuous-time, the definition of arbitrage follows the lines of its analogs in the one-step and discrete-time models. In what follows we will only consider \textit{admissible} portfolio strategies whose total value $V_t$ remains nonnegative for all times $t \in [0, T]$.

\begin{defn}{Admissible Portfolio Strategy}{AdmissiblePortfolioStrategyDef}
	A portfolio strategy $(\xi_t^{(k)})_{t \in [0, T], k=0, 1, \dots, d}$ with value
	\begin{equation} \label{eq:AdmissiblePortfolioValue}
		V_t = \xbar{\xi}_t \cdot \xbar{S}_t = \sum_{k=0}^d \xi_t^{(k)} S_t^{(k)}, \quad t \in [0, T],
	\end{equation}
	constitutes an arbitrage opportunity if all three following conditions are satisfied:
	\begin{enumerate}[label=\roman*), leftmargin=*, parsep=6pt]
		\item $V_0 \le 0$ at time $t=0$, \hfill [Start from a zero-cost portfolio, or with a debt.]
		\item $V_T \ge 0$ at time $t=T$, \hfill [Finish with a nonnegative amount.]
		\item $\Pm(V_T > 0) > 0$ at time $t=T$. \hfill [Profit is made with nonzero probability.]
	\end{enumerate}
\end{defn}

\noindent Roughly speaking, $(ii)$ means that the investor wants no loss, $(iii)$ means that he wishes to sometimes make a strictly positive gain, and $(i)$ means that he starts with zero capital or even with a debt.

Next, we turn to the definition of risk-neutral probability measures (or martingale measures) in continuous time, which states that under a risk-neutral probability measure $\Pm^*$, the return of the risky asset over the time interval $[u, t]$ equals the return of the riskless asset given by
\begin{equation} \label{eq:RisklessReturnContinuous}
	S_t^{(0)} = \e^{(t-u)r} S_u^{(0)}, \quad 0 \le u \le t.
\end{equation}
Recall that the filtration $(\cF_t)_{t \in \R_+}$ is generated by Brownian motion $(B_t)_{t \in \R_+}$, \textit{i.e.}
\begin{equation} \label{eq:BrownianFiltration}
	\cF_t = \sigma(B_u : 0 \le u \le t), \quad t \ge 0.
\end{equation}

\subsection{Self-financing Portfolio Strategies}
Let $\xi_t^{(i)}$ denote the (possibly fractional) quantity invested at time $t$ over the time interval $[t, t + \di t)$, in the asset $S_t^{(k)}$, $k = 0, 1, \dots, d$, and let
\begin{equation}
	\xbar{\xi}_t = \left(\xi_t^{(k)}\right)_{k=0,1,\dots,d}, \quad \xbar{S}_t = \left(S_t^{(k)}\right)_{k=0,1,\dots,d}, \quad t \ge 0,
\end{equation}
denote the associated portfolio value and asset price processes. The portfolio value $V_t$ at time $t \ge 0$ is given by
\begin{equation}
	V_t = \xbar{\xi}_t \cdot \xbar{S}_t = \sum_{k=0}^d \xi_t^{(k)} S_t^{(k)}, \quad t \ge 0.
	\label{eq:PortfolioValueContinuous}
\end{equation}

\begin{defn}{Self-Financing Portfolio Update}{SelfFinancingPortfolioUpdate}
	The portfolio strategy $(\xbar{\xi}_t)_{t \in \R_+}$ is self-financing if the portfolio value remains constant after updating the portfolio from $\xbar{\xi}_t$ to $\xbar{\xi}_{t+\di t}$, i.e.
	\begin{equation}
		\xbar{\xi}_t \cdot \xbar{S}_{t+\di t} = \sum_{k=0}^d \xi_t^{(k)} S_{t+\di t}^{(k)} = \sum_{k=0}^d \xi_{t+\di t}^{(k)} S_{t+\di t}^{(k)} = \xbar{\xi}_{t+\di t} \cdot \xbar{S}_{t+\di t},
		\label{eq:SelfFinancingConditionContinuous}
	\end{equation}
	which is the continuous-time analog of the self-financing condition already encountered in the discrete setting of \cref{sec:AssetsPortfoliosArbitrage}, see \ref{defn:PortfolioDef}. A major difference with the discrete-time case of \ref{defn:PortfolioDef}, however, is that the continuous-time differentials $\di S_t$ and $\di \xi_t$ do not make pathwise sense as continuous-time stochastic integrals are defined by $L^2$ limits, cf. \ref{prop:CauchyCriteriaLp}, or by convergence in probability.
\end{defn}

We start with a general model of a frictionless security market where investors are allowed to trade continuously up to some fixed finite planning horizon $T$. Uncertainty in the financial market is modelled by a probability space $(\Omega, \F, \Pm)$ and a filtration $\F = (\F_t)_{0 \le t \le T}$ satisfying the usual conditions of right-continuity and completeness. We assume that the $\sigma$-field $\F_0$ is trivial, i.e. for every $A \in \F_0$ either $\Pm(A) = 0$ or $\Pm(A) = 1$, and that $\F_T = \F$.

There are $d+1$ primary traded assets, whose price processes are given by stochastic processes $S_0, \dots, S_d$. We assume that $S = (S_0, \dots, S_d)$ follows an adapted, right-continuous with left-limits (RCLL) and strictly positive semimartingale on $(\Omega, \F, \Pm, \F)$. We assume that the processes $S_0, \dots, S_d$ represent the prices of some traded assets (stocks, bonds, or options). We will occasionally use $\cM(\Pm)$ (or just $\cM$ if reference to $\Pm$ is not needed) to denote a financial market model of the above type.

We have not emphasised so far that there was an implicit num\'{e}raire behind the prices $S_0, \dots, S_d$; it is the num\'{e}raire relevant for domestic transactions at time $t$. The formal definition of a num\'{e}raire is very much as in the discrete setting.

\begin{defn}{num\'{e}raire}{NumeraireContinuousTime}
	A num\'{e}raire is a price process $X(t)$ almost surely strictly positive for each $t \in [0, T]$.
\end{defn}

We assume now that $S_0(t)$ is a non-dividend paying asset, which is (almost surely) strictly positive and we use $S_0$ as num\'{e}raire. \emph{Historically} the money market account $B(t)$, given by $B(t) = \e^{rt}$ with a positive deterministic process $r(t)$ and $r(0) = 0$, was used as num\'{e}raire, and the reader may think of $S_0(t)$ as being $B(t)$.

Our principal task will be the pricing and hedging of contingent claims, which we model as $\F_T$-measurable random variables. This implies that the contingent claims specify a stochastic cash-flow at time $T$ and that they may depend on the whole path of the underlying in $[0,T]$, because $\F_T$ contains all that information. We will often have to impose further integrability restrictions on the contingent claims under consideration. The fundamental concept in (arbitrage) pricing and hedging contingent claims is the interplay of self-financing replicating portfolios and risk-neutral probabilities. Although the current setting is on a much higher level of sophistication, the key ideas remain the same.

We call an $\R^{d+1}$-valued predictable, locally bounded process
\[
	\varphi(t) = \bigl(\varphi_0(t),\dots,\varphi_d(t)\bigr), \quad t \in [0,T],
\]
a trading strategy (or dynamic portfolio process). (The conditions ensure that the stochastic integral $\int_{0}^{t} \varphi(u)\,\di S(u)$ exists.) Here $\varphi_i(t)$ denotes the number of shares of asset $i$ held in the portfolio at time $t$, to be determined on the basis of information available before time $t$; i.e. the investor selects his time $t$ portfolio after observing the prices $S(t_{-})$. The components $\varphi_i(t)$ may assume negative as well as positive values, reflecting the fact that we allow short sales and assume that the assets are perfectly divisible.

\begin{defn}{Trading Strategy, Value And Gains Processes}{TradingStrategyValueGains}
	\begin{enumerate}[label=(\roman*)]
		\item The value of the portfolio $\varphi$ at time $t$ is given by the scalar product
		      \begin{equation}\label{eq:ValueOfPortfolio}
			      V_{\varphi}(t) = \varphi(t) \cdot S(t) = \sum_{i=0}^{d} \varphi_i(t)\,S_i(t), \quad t \in [0,T].
		      \end{equation}
		      The process $V_{\varphi}(t)$ is called the value process, or wealth process, of the trading strategy $\varphi$.
		\item The gains process $G_{\varphi}(t)$ is defined by
		      \begin{equation}\label{eq:GainsProcess}
			      G_{\varphi}(t) = \int_{0}^{t} \varphi(u)\,\di S(u) = \sum_{i=0}^{d} \int_{0}^{t} \varphi_i(u)\,\di S_i(u).
		      \end{equation}
		\item A trading strategy $\varphi$ is called self-financing if the wealth process $V_{\varphi}(t)$ satisfies
		      \begin{equation}\label{eq:SelfFinancingCondition}
			      V_{\varphi}(t) = V_{\varphi}(0) + G_{\varphi}(t) \quad \text{for all } t \in [0,T].
		      \end{equation}
	\end{enumerate}
\end{defn}

\begin{xtip}{Interpretation of Self-Financing Condition}{InterpretationSelfFinancing}
	\begin{enumerate}[label=(\roman*)]
		\item The financial implications of the above equations are that all changes in the wealth of the portfolio are due to capital gains, as opposed to withdrawals of cash or injections of new funds.
		\item The definition of a trading strategy includes regularity assumptions in order to ensure the existence of stochastic integrals. In the current rather general setup these conditions are flexible enough to allow a discussion of the no-arbitrage condition; however, they prove too restrictive for the question of market completeness (see below).
	\end{enumerate}
\end{xtip}

Our objective is to be very flexible with regard to the chosen num\'{e}raire. The next results underline this.

\begin{prop}{Num\'{e}raire Invariance Theorem}{NumeraireInvariance}
	Self-financing portfolios remain self-financing after a num\'{e}raire change.
\end{prop}

\begin{proof}
	From a financial viewpoint this property is immediately clear. For a mathematical deduction we use the product rule to get
	\begin{equation}\label{eq:ProductRuleSkOverX}
		\di\!\left(\frac{S_k(t)}{X(t)}\right)
		= S_k(t)\,\di\!\left(\frac{1}{X(t)}\right)
		+ \frac{1}{X(t)}\,\di S_k(t)
		+ \di\!\langle S_k, 1/X \rangle (t),
	\end{equation}
	where $\langle S_k, 1/X \rangle (t)$ denotes the quadratic covariation between the semimartingales $S_k$ and $1/X$. In the same manner
	\begin{equation}\label{eq:ProductRuleVphiOverX}
		\di\!\left(\frac{V_{\varphi}(t)}{X(t)}\right)
		= V_{\varphi}(t)\,\di\!\left(\frac{1}{X(t)}\right)
		+ \frac{1}{X(t)}\,\di V_{\varphi}(t)
		+ \di\!\langle V_{\varphi}, 1/X \rangle (t).
	\end{equation}
	The self-financing condition implies that
	\begin{align}\label{eq:SelfFinancingUnderNumeraireChange}
		\di\!\left(\frac{V_{\varphi}(t)}{X(t)}\right)
		 & = \sum_{k=1}^{d} \varphi_k(t)\Bigg\{
		S_k(t)\,\di\!\left(\frac{1}{X(t)}\right)
		+ \frac{1}{X(t)}\,\di S_k(t)
		+ \di\!\langle S_k, 1/X \rangle (t)
		\Bigg\}                                                                            \\
		 & = \sum_{k=1}^{d} \varphi_k(t)\,\di\!\left(\frac{S_k(t)}{X(t)}\right). \nonumber
	\end{align}
	Thus the portfolio expressed in the new num\'{e}raire remains self-financing. \qedhere
\end{proof}

Turning back to the special num\'{e}raire \(S_0(t)\) we consider the discounted price process
\begin{equation}\label{eq:DiscountedPriceVectorByS0}
	\widetilde{S}(t) \coloneqq \frac{S(t)}{S_0(t)} = \bigl(1,\,\widetilde{S}_1(t),\dots,\widetilde{S}_d(t)\bigr),
\end{equation}
with \(\widetilde{S}_i(t) \coloneqq S_i(t)/S_0(t),\ i=1,2,\dots,d.\) Furthermore, the discounted wealth process \( \widetilde{V}_{\varphi}(t)\) is given by
\begin{equation}\label{eq:DiscountedWealthByS0}
	\widetilde{V}_{\varphi}(t) \coloneqq \frac{V_{\varphi}(t)}{S_0(t)}
	= \varphi_0(t) + \sum_{i=1}^{d} \varphi_i(t)\,\widetilde{S}_i(t).
\end{equation}

and the discounted gains process \(\widetilde{G}_{\varphi}(t)\) is
\begin{equation}\label{eq:DiscountedGainsProcess}
	\widetilde{G}_{\varphi}(t) \coloneqq \sum_{i=1}^{d} \int_{0}^{t} \varphi_i(u)\,\di \widetilde{S}_i(u).
\end{equation}
Observe that \(\widetilde{G}_{\varphi}(t)\) does not depend on the num\'{e}raire component \(\varphi_0\).

It is convenient to reformulate the self-financing condition in terms of the discounted processes:

\begin{prop}{Self-Financing In Discounted Terms}{SelfFinancingDiscounted}
	Let \(\varphi\) be a trading strategy. Then \(\varphi\) is self-financing if and only if
	\begin{equation}\label{eq:SelfFinancingDiscounted}
		\widetilde{V}_{\varphi}(t) = \widetilde{V}_{\varphi}(0) + \widetilde{G}_{\varphi}(t).
	\end{equation}
	Of course, \(V_{\varphi}(t) \ge 0\) if and only if \(\widetilde{V}_{\varphi}(t) \ge 0\).
\end{prop}

\begin{xtip}{Construction of Self-Financing Strategies}{ConstructionSelfFinancing}
	The above result shows that a self-financing strategy is completely determined by its initial value and the components \(\varphi_1, \dots, \varphi_d\). In other words, any set of predictable processes \(\varphi_1, \dots, \varphi_d\) such that the stochastic integrals \(\int \varphi_i \di \widetilde{S}_i\) exist can be uniquely extended to a self-financing strategy \(\varphi\) with specified initial value \(\widetilde{V}_{\varphi}(0)=v\) by setting

	\[
		\varphi_0(t) = v + \sum_{i=1}^{d} \int_{0}^{t} \varphi_i(u)\,\di \widetilde{S}_i(u) - \sum_{i=1}^{d} \varphi_i(t)\,\widetilde{S}_i(t), \quad t \in [0,T].
	\]
\end{xtip}

\subsection{Equivalent Martingale Measures}

We develop a relative pricing theory for contingent claims. Again the underlying concept is the link between the no-arbitrage condition and certain probability measures.

\begin{defn}{Arbitrage Opportunity}{ArbitrageOpportunity}
	A self-financing trading strategy $\varphi$ is called an arbitrage opportunity if the wealth process $V_{\varphi}$ satisfies the following set of conditions:
	\[
		V_{\varphi}(0) = 0, \quad \Pm(V_{\varphi}(T) \ge 0) = 1, \quad \text{and} \quad \Pm(V_{\varphi}(T) > 0) > 0.
	\]
\end{defn}

Arbitrage opportunities represent the limitless creation of wealth through
risk-free profit and thus should not be present in a well-functioning market.

The main tool in investigating arbitrage opportunities is the concept of
equivalent martingale measures:

\begin{defn}{Equivalent Martingale Measure}{EquivalentMartingaleMeasure}
	We say that a probability measure $\bQ$ defined on $(\Omega, \scF)$ is a (\textit{strong}) \textit{equivalent martingale measure} if:
	\begin{enumerate}[(i)]
		\item $\bQ$ is equivalent to $\Pm$,
		\item the discounted price process $\tilde{S}$ is a $\bQ$-local martingale (martingale).
	\end{enumerate}
	We denote the set of martingale measures by $\scP$.
\end{defn}

\begin{lem}{Growth Rates}{GrowthRates}
	Assume $S_0(t) = B(t) = e^{r(t)}$, then $\bQ \sim \Pm$ is a martingale measure if and only if every asset price process $S_i$ has price dynamics under $\bQ$ of the form
	\[
		dS_i(t) = r(t)S_i(t)dt + dM_i(t),
	\]
	where $M_i$ is a $\bQ$-local martingale.
\end{lem}

In order to proceed we have to impose further restrictions on the set of trading strategies.

\begin{defn}{Tame Strategy}{TameStrategy}
	A self-financing trading strategy $\varphi$ is called \textit{tame} (relative to the num\'{e}raire $S_0$) if
	\[
		\tilde{V}_{\varphi}(t) \ge 0 \quad \text{for all } t \in [0, T].
	\]

	We use the notation $\Phi$ for the set of tame trading strategies.
\end{defn}

We next analyse the value process under equivalent martingale measures for such strategies.

\begin{prop}{Value Process Martingale}{ValueProcessMartingale}
	For \(\varphi \in \Phi\), \(\tilde{V}_{\varphi}(t)\) is a non-negative local martingale, and also a supermartingale, under each \(\bQ \in \scP\).
\end{prop}

\begin{proof}
	Since $\tilde{S}(t)$ is a local martingale and $\varphi \in \Phi$ is predictable and locally bounded, it follows that the stochastic integral
	\[
		\tilde{G}_{\varphi}(t) = \int_0^t \varphi(u)d\tilde{S}(u)
	\]
	is a local martingale. Now $\tilde{V}_{\varphi}(t) = \tilde{V}_{\varphi}(0) + \tilde{G}_{\varphi}(t)$, so $\tilde{V}_{\varphi}(t)$ is a local martingale, and non-negative since $\varphi$ is tame. But non-negative local martingales are supermartingales.
\end{proof}

This observation is the key to our first central result:

\begin{thm}{Market Model Arbitrage}{MarketModelNoArbitrage}
	Assume $\scP \neq \emptyset$. Then the market model contains no arbitrage opportunities in $\Phi$.
\end{thm}

\begin{proof}
	For any $\varphi \in \Phi$ and under any $\bQ \in \scP$ $\tilde{V}_{\varphi}(t)$ is a supermartingale by \Cref{prop:ValueProcessMartingale}. That is,
	\[
		\E_{\bQ}\left(\tilde{V}_{\varphi}(t) \middle| \scF_u\right) \le \tilde{V}_{\varphi}(u), \text{ for all } u \le t \le T.
	\]
	For $\varphi \in \Phi$ to be an arbitrage opportunity we must have $\tilde{V}_{\varphi}(0) = V_{\varphi}(0) = 0$. So
	\[
		\E_{\bQ}\left(\tilde{V}_{\varphi}(t)\right) \le 0, \text{ for all } 0 \le t \le T.
	\]
	Now $\varphi$ is tame, so $\tilde{V}_{\varphi}(t) \ge 0$, $0 \le t \le T$, implying $\E_{\bQ}\left(\tilde{V}_{\varphi}(t)\right) = 0$, $0 \le t \le T$, and in particular $\E_{\bQ}\left(\tilde{V}_{\varphi}(T)\right) = 0$. But an arbitrage opportunity $\varphi$ also has to satisfy $\Pm(V_{\varphi}(T) \ge 0) = 1$, and since $\bQ \sim \Pm$, this means $\bQ(V_{\varphi}(T) \ge 0) = 1$. Both together yield
	\[
		\bQ(V_{\varphi}(T) > 0) = \Pm(V_{\varphi}(T) > 0) = 0,
	\]
	and hence the result follows.
\end{proof}

The following fundamental theorem of asset pricing is proved in [55].

\begin{thm}{Fundamental Theorem of Asset Pricing}{FundamentalTheoremOfAssetPricing}
	There exists an equivalent martingale measure for the financial market model $\cM(\Pm)$ if and only if the condition \textit{NFLVR} holds true.
\end{thm}

\subsection{Risk-neutral Pricing}

We now assume that there exists a strong equivalent martingale measure $\Pm^*$ (which implies that there are no arbitrage opportunities with respect to $\Phi$ in the financial market model $\cM$.) Until further notice we use $\Pm^*$ as our reference measure, and when using the terms martingale and local martingale we always assume that the underlying probability measure is $\Pm^*$. In particular, we restrict our attention to contingent claims $X$ such that $X/S_0(T) \in L^1(\scF, \Pm^*)$.

We now define a further subclass of trading strategies:

\begin{defn}{Admissible Strategy}{AdmissibleStrategy}
	A self-financing trading strategy $\varphi$ is called \textit{($\Pm^*$-) admissible} if the relative gains process
	\[
		\tilde{G}_{\varphi}(t) = \int_0^t \varphi(u)d\tilde{S}(u)
	\]
	is a ($\Pm^*$-) martingale. The class of all ($\Pm^*$-) admissible trading strategies is denoted $\Phi(\Pm^*)$.
\end{defn}

By definition $\tilde{S}$ is a martingale, and $\tilde{G}$ is the stochastic integral with respect to $\tilde{S}$. By \Cref{xtip:ConstructionSelfFinancing} we see that any sufficiently integrable processes $\varphi_1, \dots, \varphi_d$ give rise to $\Pm^*$-admissible trading strategies.

We do not assume that admissible trading strategies are tame. However, since we only used the supermartingale property of \(\tilde{G}_{\varphi}\) in the proof of \Cref{thm:MarketModelNoArbitrage}, we can repeat the above argument to obtain

\begin{thm}{No Arbitrage for Admissible Strategies}{NoArbitrageForAdmissibleStrategies}
	The financial market model $\cM$ contains no arbitrage opportunities in $\Phi(\Pm^*)$.
\end{thm}

Under the assumption that no arbitrage opportunities exist, the question of pricing and hedging a contingent claim reduces to the existence of replicating self-financing trading strategies. Formally:

\begin{defn}{Attainable Claims and Market Completeness}{AttainableClaimsAndMarketCompleteness}
	\begin{enumerate}[label=(\roman*), leftmargin=*]
		\item A contingent claim $X$ is called \textit{attainable} if there exists an admissible trading strategy $\varphi \in \Phi(\Pm^*)$ such that
		      \[
			      V_{\varphi}(T) = X.
		      \]
		      Such a $\varphi$ is called a \textit{replicating strategy} for $X$.
		\item The financial market model $\cM$ is \textit{complete} if every contingent claim is attainable.
	\end{enumerate}
\end{defn}

Again we emphasise that this depends on the class of trading strategies. On the other hand, it does not depend on the num\'{e}raire: it is an easy exercise to show that if a contingent claim is attainable in a given num\'{e}raire it is also attainable in any other num\'{e}raire and the replicating strategies are the same.

If a contingent claim $X$ is attainable, $X$ can be replicated by a portfolio $\varphi \in \Phi(\Prob^*)$. This means that holding the portfolio and holding the contingent claim are equivalent from a financial point of view. In the absence of arbitrage the (arbitrage) price process $\Pi_X(t)$ of the contingent claim must therefore satisfy
\[
	\Pi_X(t) = V_{\varphi}(t).
\]
Of course the questions arise of what will happen if $X$ can be replicated by more than one portfolio, and what the relation of the price process to the equivalent martingale measure(s) is. The following central theorem is the key to answering these questions:

\begin{thm}{Risk-Neutral Valuation Formula}{RiskNeutralValuationFormula}
	The arbitrage price process of any attainable claim is given by the risk-neutral valuation formula
	\begin{equation}
		\Pi_X(t) = S_0(t)\E_{\Prob^*} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right].
	\end{equation}
\end{thm}

\begin{proof}
	Since $X$ is attainable, there exists a replicating strategy $\varphi \in \Phi(\Prob^*)$ such that $V_{\varphi}(T) = X$ and $\Pi_X(t) = V_{\varphi}(t)$. Since $\varphi \in \Phi(\Prob^*)$ the discounted value process $\tilde{V}_{\varphi}(t)$ is a martingale, and hence
	\begin{align*}
		\Pi_X(t) & = V_{\varphi}(t) = S_0(t)\tilde{V}_{\varphi}(t)                                                                                                       \\
		         & = S_0(t)\E_{\Prob^*} \left[ \tilde{V}_{\varphi}(T) \Big| \cF_t \right] = S_0(t)\E_{\Prob^*} \left[ \frac{V_{\varphi}(T)}{S_0(T)} \Bigg| \cF_t \right] \\
		         & = S_0(t)\E_{\Prob^*} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right].
	\end{align*}
\end{proof}

The uniqueness question is immediate from the above theorem:

\begin{cor}{Uniqueness of Replicating Portfolio Value}{UniquenessOfReplicatingPortfolioValue}
	For any two replicating portfolios $\varphi, \psi \in \Phi(\Prob^*)$ we have
	\[
		V_{\varphi}(t) = V_{\psi}(t).
	\]
\end{cor}

\begin{xattention}{Choice of Martingale Measure}{ChoiceOfMartingaleMeasure}
	The above argument addresses only the case of different replicating portfolios for $X$ in $\Phi(\Prob^*)$. Of course the choice of $\Prob^*$ was itself completely arbitrary, and we could have several - indeed, many - equivalent martingale measures $\bQ$. This makes the definition of admissible strategies with respect to different martingale measures somewhat unsatisfactory. However, if we combine our previous two definitions and call a self-financing trading strategy $\varphi$ admissible, if $\varphi$ is tame and $\bQ$-admissible for some $\bQ \in \scP$, and $X$ attainable, if there exists an admissible $\varphi$ replicating it, we show that for admissible $\varphi_1 \in \Phi(\bQ_1)$ and $\varphi_2 \in \Phi(\bQ_2)$ replicating $X$ we have
	\[
		\E_{\bQ_1} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right] = \E_{\bQ_2} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right],
	\]
	implying that the arbitrage price process is unique. More generally, one can even show that
	\[
		\Pi_X(0) = \sup_{\bQ \in \scP} S_0(0)\E_{\bQ} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right] = \inf_{\varphi \in \theta(X)} V_{\varphi}(0),
	\]
	where $\theta(X)$ is the class of all admissible (in the wider sense) trading strategies replicating $X$.
\end{xattention}

\begin{xtip}{Existence of Equivalent Martingale Measures}{ExistenceOfEMM}
	The question of existence of (strong) equivalent martingale measures for a given stochastic process $S$ in our market model is closely related to properties of the stochastic (Doléans) exponential $\mathcal{E}(U)$. Loosely speaking, a probability measure $\bQ$ on $(\Omega, \cF_T)$ is equivalent to the underlying probability measure $\Prob$ if the Radon-Nikodým derivative $d\bQ/d\Prob$ is of the form $\mathcal{E}(U)$ for some local martingale $U$.
\end{xtip}

For pure pricing purposes it is sufficient to find a strong equivalent martingale measure, but for risk-management purposes it is more important to find the replicating portfolio. We have the following useful result, which shows that hedging is equivalent to the existence of a stochastic integral representation of the discounted claim.

\begin{lem}{Integral Representation Implies Attainability}{IntegralRepImpliesAttainability}
	Assume that the discounted contingent claim $X/S_0(T)$ is $\Prob^*$-integrable. If the $(\Prob^*)$-martingale $M$ defined by
	\[
		M(t) = \E_{\Prob^*} \left[ \frac{X}{S_0(T)} \Bigg| \cF_t \right]
	\]
	admits an integral representation of the form
	\[
		M(t) = x + \sum_{i=1}^d \int_0^t \varphi_i(u) \di \tilde{S}_i(u),
	\]
	with $\varphi_1, \dots, \varphi_d$ predictable and locally bounded, then $X$ is attainable.
\end{lem}

\begin{proof}
	We are looking for a trading strategy $\varphi \in \Phi(\Prob^*)$ such that $V_{\varphi}(T) = X$. We use the components $\varphi_1, \dots, \varphi_d$ from the definition of the integral representation of $M$ and then define $\varphi_0$ as in \Cref{xtip:ConstructionSelfFinancing}, with $v=x$. Then $\varphi = (\varphi_0, \varphi_1, \dots, \varphi_d)$ is a self-financing trading strategy with
	\[
		\tilde{V}_{\varphi}(t) = v + \tilde{G}_{\varphi}(t) = x + \sum_{i=1}^d \int_0^t \varphi_i(u) \di \tilde{S}_i(u) = M(t).
	\]
	By definition $M$ is a non-negative martingale, so $\varphi \in \Phi(\Prob^*)$ follows, and also by definition
	\[
		V_{\varphi}(T) = S_0(T)\tilde{V}_{\varphi}(T) = S_0(T)M(T) = S_0(T)\frac{X}{S_0(T)} = X.
	\]
\end{proof}

We thus see that (given some integrability conditions) attainability, and hence completeness, is closely linked to martingale representation theorems. Statement and proofs of the martingale representation property require rather technical definitions involving the fine structure of the filtration $\cF$. Moreover, in a general set-up the conditions on the process $\varphi$ (the integrand) no longer guarantee that $\varphi$ may be used as a suitable trading strategy.

We close by stating the following partial analogue of the completeness theorem in the discrete setting:

\begin{thm}{Restricted Completeness}{RestrictedCompleteness}
	If the strong martingale measure $\Prob^*$ is the unique martingale measure for the financial market model $\cM$, then $\cM$ is complete, in the restricted sense that every contingent claim $X$ satisfying
	\[
		\frac{X}{S_0(T)} \in L^1(\cF, \Prob^*)
	\]
	is attainable.
\end{thm}

\subsection{Change of Num\'{e}raire}

We now develop the technique of change of num\'{e}raire, which will be extremely useful in the valuation of contingent claims, and also for deduction of theoretical results.

We put ourselves in the following framework. Assume that we used \(S_0(t)\) as num\'{e}raire, and that there exists an equivalent martingale measure \(\Prob^*\).

\begin{thm}{Numéraire Change Measure}{NumeraireChangeMeasure}
	Let $X(t)$ be a non-dividend-paying num\'{e}raire such that $X(t)/S_0(t)$ is a $\Prob^*$-martingale. Then there exists a probability measure $\bQ_X$, defined by its Radon-Nikodým derivative $\eta(T)$ with respect to $\Prob^*$,
	\[
		\eta(t) = \frac{\di \bQ_X}{\di \Prob^*} \bigg|_{\cF_t} = \frac{X(t)}{X(0)S_0(t)},
	\]
	such that:
	\begin{enumerate}[label=(\roman*), ref=(\roman*), leftmargin=*]
		\item the basic security prices discounted with respect to $X$ are $\bQ_X$-local martingales;
		\item if a contingent claim $Y$ is attainable under $(S_0(t), \Prob^*)$, then it is attainable under $(X(t), \bQ_X)$, and the replicating portfolio is the same. So the arbitrage price processes, given by the risk-neutral valuation formula, coincide.
	\end{enumerate}
\end{thm}

\begin{proof}
	\begin{enumerate}[label=(\roman*), leftmargin=*]
		\item Denote by $\bar{Z} = Z(t)/X(t)$ the relative price of a security $Z$ (it may be one of the components $S_i$) with respect to the num\'{e}raire $X(t)$. Since $\eta(t) = X(t)/X(0)S_0(t)$ is a $\Prob^*$-martingale we can use the Bayes formula and the $\Prob^*$-martingale property of $Z(t)/S_0(t)$ to get for $0 \le u \le t$
		      \begin{align*}
			      \E_{\bQ_X} \left[ \bar{Z}(t) \middle| \cF_u \right]
			       & = \frac{1}{\eta(u)} \E_{\Prob^*} \left[ \bar{Z}(t)\eta(t) \middle| \cF_u \right]                             \\
			       & = \frac{X(0)S_0(u)}{X(u)} \E_{\Prob^*} \left[ \frac{Z(t)}{X(t)} \frac{X(t)}{X(0)S_0(t)} \Bigg| \cF_u \right] \\
			       & = \frac{S_0(u)}{X(u)} \frac{Z(u)}{S_0(u)} = \frac{Z(u)}{X(u)} = \bar{Z}(u).
		      \end{align*}
		      So the basic securities are martingales, and hence so is any portfolio.
		\item If $Y$ is attainable under $\Prob^*$, then there exists a $\Prob^*$-admissible portfolio $\varphi$ such that $V_{\varphi}(T) = Y$, and by the risk-neutral valuation formula
		      \[
			      V_{\varphi}(t) = S_0(t)\E_{\Prob^*} \left[ \frac{Y}{S_0(T)} \Bigg| \cF_t \right].
		      \]
		      Using again the Bayes formula we obtain
		      \begin{align*}
			      \E_{\bQ_X} \left[ \frac{Y}{X(T)} \Bigg| \cF_t \right]
			       & = \frac{1}{\eta(t)} \E_{\Prob^*} \left[ \frac{Y}{X(T)} \eta(T) \Bigg| \cF_t \right]                       \\
			       & = \frac{X(0)S_0(t)}{X(t)} \E_{\Prob^*} \left[ \frac{Y}{X(T)} \frac{X(T)}{X(0)S_0(T)} \Bigg| \cF_t \right] \\
			       & = \frac{V_{\varphi}(t)}{X(t)}.
		      \end{align*}
		      Since self-financing portfolios remain self-financing under a change of num\'{e}raire (by the num\'{e}raire invariance theorem), and since by part (i) $V_{\varphi}(t)$ is a $\bQ_X$-martingale, it follows that $\varphi$ is $\bQ_X$-admissible and the result follows.
	\end{enumerate}
\end{proof}

\begin{cor}{Change of Numéraire Expectation Formula}{ChangeOfNumeraireExpectation}
	Let $X(t), Y(t)$ be numéraires satisfying the assumptions of the above proposition and $Z$ a contingent claim. Then we have the change of numéraire formula
	\begin{equation}\label{eq:ChangeOfNumeraireExpectationFormula}
		X(t)\E_{\bQ_X} \left[ \frac{Z}{X(T)} \Bigg| \cF_t \right] = Y(t)\E_{\bQ_Y} \left[ \frac{Z}{Y(T)} \Bigg| \cF_t \right].
	\end{equation}
\end{cor}

\begin{proof}
	We can use the second part of the above proposition, or use the Radon-Nikodým derivative
	\[
		\frac{\di \bQ_X}{\di \bQ_Y} \bigg|_{\cF_T} = \frac{\di \bQ_X}{\di \Prob^*} \bigg/ \frac{\di \bQ_Y}{\di \Prob^*} = \frac{X(T)}{X(0)S_0(T)} \bigg/ \frac{Y(T)}{Y(0)S_0(T)} = \frac{X(T)/Y(T)}{X(0)/Y(0)}
	\]
	and repeat the argument.
\end{proof}

The main application of changes of num\'{e}raire is its usefulness in pricing contingent claims. In short, we claim there is a best num\'{e}raire for any given problem, and it pays to find it and work with it.

\begin{prop}{Option Pricing under Numéraire Change}{OptionPricingNumeraireChange}
	Denote by $T$ and $K$ respectively the maturity and exercise price of a European call option on an asset $Z$. The time $t=0$ price $C(0)$ of the call can be written as
	\[
		\frac{C(0)}{p(0,T)} = \E_{\bQ^T} \left[ (Z(T) - K)^+ \right],
	\],
	or
	\[
		C(0) = Z(0)\bQ_Z(A) - Kp(0,T)\bQ^T(A),
	\]
	where $A = \{ \omega : Z(T, \omega) > K \}$.
\end{prop}

\begin{proof}
	The first expression is immediate from \Cref{thm:NumeraireChangeMeasure}. To derive the second write
	\[
		\frac{C(0)}{p(0,T)} = \E_{\bQ^T} \left[ (Z(T) - K)^+ \right] = \E_{\bQ^T} \left[ Z(T)\I_A \right] - K\bQ^T(A).
	\]
	Using $Z$ as a numéraire in the middle expectation, we have by the change of numéraire, \Cref{eq:ChangeOfNumeraireExpectationFormula}
	\[
		p(0,T)\E_{\bQ^T} \left[ \frac{Z(T)}{p(T,T)} \I_A \right] = Z(0)\E_Z \left[ \I_A \right],
	\]
	which gives the second expression.
\end{proof}
